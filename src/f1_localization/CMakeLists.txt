cmake_minimum_required(VERSION 3.8)
project(f1_localization)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(cartographer REQUIRED)
find_package(cartographer_ros REQUIRED)
add_executable(gnss_cartographer src/gnss_cartographer.cpp)
target_compile_features(gnss_cartographer PRIVATE cxx_std_17)
ament_target_dependencies(gnss_cartographer cartographer_ros)
target_link_libraries(gnss_cartographer cartographer)
add_executable(read_gnss_reference src/read_gnss_reference.cpp)
target_link_libraries(read_gnss_reference PRIVATE cartographer)
install(TARGETS read_gnss_reference gnss_cartographer DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/localization_manager DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(localization test/test_localization.py)
  ament_add_pytest_test(map_io test/test_map_io.py)
  ament_add_pytest_test(launch test/test_launch.py)
  ament_add_pytest_test(manager test/test_manager.py TIMEOUT 45)
  ament_add_pytest_test(gnss_slam test/test_gnss_slam.py TIMEOUT 110)
  ament_add_pytest_test(open_water test/test_open_water.py TIMEOUT 80)
endif()
ament_package()
