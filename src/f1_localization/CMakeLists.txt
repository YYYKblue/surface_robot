cmake_minimum_required(VERSION 3.8)
project(f1_localization)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/localization_manager DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(localization test/test_localization.py)
  ament_add_pytest_test(map_io test/test_map_io.py)
  ament_add_pytest_test(launch test/test_launch.py)
  ament_add_pytest_test(manager test/test_manager.py TIMEOUT 45)
endif()
ament_package()
