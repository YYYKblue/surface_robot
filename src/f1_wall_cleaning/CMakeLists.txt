cmake_minimum_required(VERSION 3.8)
project(f1_wall_cleaning)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/wall_perception scripts/wall_controller scripts/wall_demo_scan DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(perception test/test_perception.py)
  ament_add_pytest_test(control test/test_control.py)
  ament_add_pytest_test(ros test/test_ros.py TIMEOUT 45)
endif()
ament_package()
