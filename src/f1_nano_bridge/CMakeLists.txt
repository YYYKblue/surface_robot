cmake_minimum_required(VERSION 3.8)
project(f1_nano_bridge)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
# Different Python name prevents collision with the legacy top-level nano_bridge.
ament_python_install_package(nano_ros_bridge)
install(PROGRAMS scripts/f1_bridge scripts/imu_bridge scripts/f1_ui_gateway
        DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md IMU.md WINDOWS_UI.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(protocol test/test_protocol.py)
  ament_add_pytest_test(session test/test_session.py)
  ament_add_pytest_test(ui_protocol test/test_ui_protocol.py TIMEOUT 30)
  ament_add_pytest_test(ui_gateway test/test_ui_gateway.py TIMEOUT 30)
  ament_add_pytest_test(firmware test/test_firmware.py TIMEOUT 60)
  ament_add_pytest_test(transport test/test_transport.py TIMEOUT 30)
  ament_add_pytest_test(ros test/test_ros.py TIMEOUT 30)
  ament_add_pytest_test(imu test/test_imu.py)
  ament_add_pytest_test(imu_transport test/test_imu_transport.py TIMEOUT 30)
endif()
ament_package()
