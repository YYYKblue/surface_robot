cmake_minimum_required(VERSION 3.8)
project(f1_l_cleaning)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/l_controller DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(route test/test_route.py)
  ament_add_pytest_test(mission test/test_mission.py)
  ament_add_pytest_test(ros test/test_ros.py TIMEOUT 60)
  ament_add_pytest_test(bringup test/test_bringup.py TIMEOUT 60)
endif()
ament_package()
