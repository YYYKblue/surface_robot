cmake_minimum_required(VERSION 3.8)
project(f1_area_patrol)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/area_patrol scripts/patrol_preview DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(patrol test/test_patrol.py)
  ament_add_pytest_test(patrol_ros test/test_ros_patrol.py TIMEOUT 60)
endif()
ament_package()
