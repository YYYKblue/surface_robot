cmake_minimum_required(VERSION 3.8)
project(f1_rtk_driver)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)

ament_python_install_package(wit_rtk)
install(PROGRAMS scripts/rtk_driver scripts/configure_ntrip DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config deploy DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(nmea test/test_nmea.py)
  ament_add_pytest_test(discovery test/test_discovery.py)
  ament_add_pytest_test(ntrip test/test_ntrip.py TIMEOUT 60)
  ament_add_pytest_test(serial_launch test/test_serial_launch.py TIMEOUT 60)
endif()
ament_package()
