cmake_minimum_required(VERSION 3.8)
project(f1_mapping_bringup)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(cartographer REQUIRED)
find_package(cartographer_ros REQUIRED)
find_package(cartographer_ros_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(f1_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
# Override the matching upstream frontend objects in this executable only.
get_target_property(cartographer_type cartographer TYPE)
set(cartographer_manifest "")
if(EXISTS "${cartographer_DIR}/package.xml")
  file(READ "${cartographer_DIR}/package.xml" cartographer_manifest)
endif()
set(heading_sources src/wt102_heading.cpp src/lake_pose.cpp)
if(cartographer_type STREQUAL "STATIC_LIBRARY" AND
   cartographer_manifest MATCHES "<version>2[.]0[.]9004</version>")
  set(wt102_supported TRUE)
  list(APPEND heading_sources src/cartographer_2_0_9004/pose_extrapolator.cc
    src/cartographer_2_0_9004/cartographer/mapping/internal/2d/local_trajectory_builder_2d.cc
    src/cartographer_2_0_9004/cartographer/mapping/2d/submap_2d.cc)
else()
  message(STATUS "WT102 prediction unavailable for this Cartographer version; building standard mapping")
endif()
add_executable(gnss_cartographer src/gnss_cartographer.cpp ${heading_sources})
target_compile_definitions(gnss_cartographer PRIVATE F1_WT102_SUPPORTED=$<BOOL:${wt102_supported}>)
target_include_directories(gnss_cartographer PRIVATE src src/cartographer_2_0_9004)
target_compile_features(gnss_cartographer PRIVATE cxx_std_17)
ament_target_dependencies(gnss_cartographer f1_interfaces cartographer_ros cartographer_ros_msgs
  geometry_msgs nav_msgs rclcpp sensor_msgs tf2_ros)
target_link_libraries(gnss_cartographer cartographer)
install(TARGETS gnss_cartographer DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/scan_check scripts/gnss_positions DESTINATION lib/${PROJECT_NAME})
install(FILES src/cartographer_2_0_9004/LICENSE DESTINATION share/${PROJECT_NAME}/cartographer)
install(FILES README.md README_LIDAR.md CARTOGRAPHER_PRINCIPLES.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  if(wt102_supported)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(wt102_prediction test/test_wt102_prediction.cpp src/wt102_heading.cpp src/lake_pose.cpp src/cartographer_2_0_9004/pose_extrapolator.cc)
    target_include_directories(wt102_prediction PRIVATE src src/cartographer_2_0_9004)
    target_compile_features(wt102_prediction PRIVATE cxx_std_17)
    target_link_libraries(wt102_prediction cartographer)
    ament_add_pytest_test(wt102_slam test/test_wt102_slam.py TIMEOUT 90)
  endif()
  ament_add_pytest_test(configuration test/test_configuration.py)
  ament_add_pytest_test(lidar test/test_lidar.py)
  ament_add_pytest_test(gnss test/test_gnss.py)
endif()

ament_package()
