cmake_minimum_required(VERSION 3.8)
project(f1_rectangle_cleaning)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/rectangle_controller scripts/rectangle_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DEPLOY_TEST_NANO.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(mission test/test_mission.py)
  ament_add_pytest_test(heading_feedback test/test_heading_feedback.py)
  ament_add_pytest_test(ros test/test_ros.py TIMEOUT 45)
endif()
ament_package()
